# Three layered sounds: an impact, a magic sparkle and a door creak.
# Run it:  soundclaw new demo.sclaw && soundclaw run demo.sclaw docs/examples/layered-demo.txt

# an impact: a low thump that falls in pitch, a burst of dulled noise, and a click
sound new impact voice=layered seed=4
layer add impact osc wave=sine freq=140 dur=0.5 gain=0.9
env impact 1.freq "0:140 0.25:42@exp"
env impact 1.amp adsr(0.002,0.28,0,0)
layer add impact osc wave=brown dur=0.2 filter=lp cutoff=2500 gain=0.8
env impact 2.cutoff "0:5000 0.15:300@exp"
env impact 2.amp "0:0 0.002:1 0.18:0@exp"
layer add impact osc wave=square freq=3200 dur=0.012 gain=0.3
env impact 3.amp "0:0 0.001:1 0.01:0@exp"
fx add impact distort shape=soft drive=5 mix=0.7
fx add impact reverb size=0.25 mix=0.12
sound set impact gain=2
fx add impact limit ceiling=-1 release=40 lookahead=3   # loud and not clipping: limit the peaks, then the gain can go up

# a magic sparkle: four bell notes spread across the stereo field, a shimmer, echo and room
sound new sparkle voice=layered seed=12
layer add sparkle fm freq=1568 ratio2=3.5 index2=4 dur=0.6 gain=0.5 pan=-0.6
env sparkle 1.index2 "0:5 0.4:0@exp"
env sparkle 1.amp "0:0 0.004:1 0.5:0@exp"
layer add sparkle fm freq=1976 ratio2=3.5 index2=4 dur=0.6 gain=0.45 pan=0.4 offset=0.06
env sparkle 2.index2 "0:5 0.4:0@exp"
env sparkle 2.amp "0:0 0.004:1 0.5:0@exp"
layer add sparkle fm freq=2349 ratio2=3.5 index2=4 dur=0.6 gain=0.4 pan=-0.2 offset=0.12
env sparkle 3.index2 "0:5 0.4:0@exp"
env sparkle 3.amp "0:0 0.004:1 0.5:0@exp"
layer add sparkle fm freq=3136 ratio2=3.5 index2=4 dur=0.6 gain=0.35 pan=0.7 offset=0.18
env sparkle 4.index2 "0:5 0.4:0@exp"
env sparkle 4.amp "0:0 0.004:1 0.5:0@exp"
layer add sparkle osc wave=white dur=0.5 filter=hp cutoff=7000 gain=0.1
env sparkle 5.amp "0:0 0.1:1 0.45:0@exp"
fx add sparkle delay time=0.14 feedback=0.35 mix=0.25
fx add sparkle reverb size=0.6 damp=0.4 mix=0.25
sound set sparkle gain=1.25

# a door creak: a saw whose pitch bends, through a moving band-pass and a little distortion, over a dull breath
sound new creak voice=layered seed=3
layer add creak osc wave=saw dur=1.05 filter=bp cutoff=700 q=7 gain=0.6
env creak 1.freq "0:150 0.35:260 0.7:210 1.05:130"
env creak 1.cutoff "0:500 1.05:1400"
env creak 1.amp "0:0 0.15:1 0.6:0.8 1.05:0@exp"
fx add creak layer=1 distort shape=soft drive=5 mix=0.5
layer add creak osc wave=pink dur=1.05 filter=lp cutoff=900 gain=0.25 seed=4
env creak 2.amp "0:0 0.2:1 1.05:0@exp"
sound set creak gain=1.6
